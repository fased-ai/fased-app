package main

import (
	"context"
	"encoding/json"
	"fmt"
	"os"
	"os/signal"
	"syscall"
	"time"

	"github.com/fased-ai/fased-app/internal/core"
)

// Disposable schema33 core/IPC fixture, never an installer or live connector.
func main() {
	if os.Getuid() == 0 || len(os.Args) != 2 {
		panic("ordinary fixture owner and data directory required")
	}
	ctx, cancel := signal.NotifyContext(context.Background(), syscall.SIGTERM, os.Interrupt)
	defer cancel()
	server, err := core.Start(ctx, os.Args[1], "127.0.0.1:0", "", nil)
	if err != nil {
		panic(err)
	}
	code, err := server.Store.IssueClaimCode(ctx)
	if err != nil {
		panic(err)
	}
	if err = json.NewEncoder(os.Stdout).Encode(map[string]string{"code": code.Code}); err != nil {
		panic(err)
	}
	go func() {
		<-ctx.Done()
		closeCtx, done := context.WithTimeout(context.Background(), 5*time.Second)
		defer done()
		server.Close(closeCtx)
	}()
	if err = server.Serve(); err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
}
